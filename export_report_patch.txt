    def export_report(self, fmt):
        if not self.current_results:
            QMessageBox.warning(self, "Нет данных", "Сначала выполните сканирование.")
            return
        
        ext_map = {'json': 'json', 'html': 'html', 'pdf': 'pdf'}
        file_path, _ = QFileDialog.getSaveFileName(
            self, f"Сохранить отчёт в {fmt.upper()}",
            f"scan_report.{ext_map[fmt]}",
            f"{fmt.upper()} files (*.{ext_map[fmt]})"
        )
        if not file_path:
            return
        
        try:
            if fmt == 'json':
                export_json(self.current_results, file_path)
            elif fmt == 'html' or fmt == 'pdf':
                # Используем AdvancedReportGenerator для красивых отчётов
                from advanced_report import AdvancedReportGenerator
                import sqlite3, tempfile, os
                
                # Создаём временную БД с данными
                temp_db = tempfile.NamedTemporaryFile(delete=False, suffix='.db')
                temp_db.close()
                conn = sqlite3.connect(temp_db.name)
                c = conn.cursor()
                c.execute('''CREATE TABLE vulnerabilities (
                    id INTEGER PRIMARY KEY,
                    severity TEXT,
                    description TEXT,
                    recommendation TEXT
                )''')
                for item in self.current_results:
                    c.execute('INSERT INTO vulnerabilities (severity, description, recommendation) VALUES (?, ?, ?)',
                              (item.get('severity', 'UNKNOWN'), item.get('description', ''), item.get('recommendation', '')))
                conn.commit()
                conn.close()
                
                generator = AdvancedReportGenerator(temp_db.name, logo_path="logo.png")
                if fmt == 'html':
                    generator.generate_html(file_path)
                else:
                    generator.generate_pdf(file_path)
                
                os.unlink(temp_db.name)
            
            QMessageBox.information(self, "Успех", f"Отчёт сохранён как {file_path}")
        except Exception as e:
            QMessageBox.critical(self, "Ошибка", f"Не удалось сохранить: {e}")
